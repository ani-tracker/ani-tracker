cmake_minimum_required(VERSION 3.24)

project(ani_torrent_tauri_android LANGUAGES CXX)

if(NOT DEFINED ANI_ANDROID_NATIVE_ROOT)
  message(FATAL_ERROR "ANI_ANDROID_NATIVE_ROOT is required")
endif()

if(ANDROID_ABI STREQUAL "arm64-v8a")
  set(ANI_ANDROID_TRIPLET "arm64-android")
elseif(ANDROID_ABI STREQUAL "x86_64")
  set(ANI_ANDROID_TRIPLET "x64-android")
else()
  message(FATAL_ERROR "Unsupported Ani torrent Android ABI: ${ANDROID_ABI}")
endif()

set(ANI_ANDROID_NATIVE_PREFIX "${ANI_ANDROID_NATIVE_ROOT}/${ANI_ANDROID_TRIPLET}")
set(ANI_BUILD_SIDECAR OFF CACHE BOOL "" FORCE)
set(ANI_BUILD_ANDROID_JNI ON CACHE BOOL "" FORCE)
set(ANI_FETCH_LIBTORRENT ON CACHE BOOL "" FORCE)
set(BOOST_ROOT "${ANI_ANDROID_NATIVE_PREFIX}" CACHE PATH "" FORCE)
set(Boost_INCLUDE_DIR "${ANI_ANDROID_NATIVE_PREFIX}/include" CACHE PATH "" FORCE)
set(OPENSSL_ROOT_DIR "${ANI_ANDROID_NATIVE_PREFIX}" CACHE PATH "" FORCE)
set(OPENSSL_USE_STATIC_LIBS TRUE CACHE BOOL "" FORCE)

add_subdirectory(
  "${CMAKE_CURRENT_LIST_DIR}/../../../../../../native/torrent-core"
  "${CMAKE_CURRENT_BINARY_DIR}/torrent-core"
)
